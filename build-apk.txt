          package com.nadiayousuf.stepbyn

          import android.app.*
          import android.content.Context
          import android.content.Intent
          import android.content.pm.ServiceInfo
          import android.hardware.Sensor
          import android.hardware.SensorEvent
          import android.hardware.SensorEventListener
          import android.hardware.SensorManager
          import android.os.Build
          import android.os.IBinder

          /** Foreground service that listens to the hardware step counter 24/7, even when the app UI is closed. */
          class StepService : Service(), SensorEventListener {
              private lateinit var sm: SensorManager
              private var lastNotify = 0L

              override fun onBind(intent: Intent?): IBinder? = null

              override fun onCreate() {
                  super.onCreate()
                  val nm = getSystemService(NotificationManager::class.java)
                  nm.createNotificationChannel(NotificationChannel(CH, "Step counting", NotificationManager.IMPORTANCE_LOW))
                  val n = notification(Store.steps(this))
                  if (Build.VERSION.SDK_INT >= 34) startForeground(ID, n, ServiceInfo.FOREGROUND_SERVICE_TYPE_HEALTH)
                  else startForeground(ID, n)
                  sm = getSystemService(Context.SENSOR_SERVICE) as SensorManager
                  sm.getDefaultSensor(Sensor.TYPE_STEP_COUNTER)?.let {
                      sm.registerListener(this, it, SensorManager.SENSOR_DELAY_NORMAL, 30_000_000)
                  }
              }

              override fun onStartCommand(intent: Intent?, flags: Int, startId: Int) = START_STICKY

              override fun onSensorChanged(e: SensorEvent) {
                  val steps = Store.onSensorTotal(this, e.values[0].toLong())
                  val now = System.currentTimeMillis()
                  if (now - lastNotify > 10_000) {
                      lastNotify = now
                      getSystemService(NotificationManager::class.java).notify(ID, notification(steps))
                  }
              }

              override fun onAccuracyChanged(s: Sensor?, a: Int) {}

              override fun onDestroy() {
                  if (::sm.isInitialized) sm.unregisterListener(this)
                  super.onDestroy()
              }

              private fun notification(steps: Int): Notification {
                  val pi = PendingIntent.getActivity(this, 0, Intent(this, MainActivity::class.java), PendingIntent.FLAG_IMMUTABLE)
                  return Notification.Builder(this, CH)
                      .setSmallIcon(R.drawable.ic_stat_step)
                      .setContentTitle("Step by N")
                      .setContentText("%,d steps today".format(steps))
                      .setContentIntent(pi)
                      .setOngoing(true)
                      .build()
              }

              companion object { const val CH = "steps"; const val ID = 1 }
          }
          SBN_EOF

          package com.nadiayousuf.stepbyn

          import android.content.Context
          import java.time.LocalDate

          /** On-device storage of daily step totals. */
          object Store {
              private fun sp(c: Context) = c.applicationContext.getSharedPreferences("sbn_steps", Context.MODE_PRIVATE)
              fun today(): String = LocalDate.now().toString()
              fun steps(c: Context, date: String = today()): Int = sp(c).getInt("s_$date", 0)

              fun lastSevenDays(c: Context): List<Pair<String, Int>> {
                  val t = LocalDate.now()
                  return (6 downTo 0).map { val d = t.minusDays(it.toLong()).toString(); d to steps(c, d) }
              }

              /** Receives the sensor running total. Handles first run and phone reboots (counter resets to 0). */
              @Synchronized
              fun onSensorTotal(c: Context, total: Long): Int {
                  val s = sp(c)
                  val last = s.getLong("last", -1L)
                  val delta = when {
                      last < 0 -> 0L
                      total >= last -> total - last
                      else -> total
                  }
                  val key = "s_" + today()
                  val now = s.getInt(key, 0) + delta.toInt()
                  s.edit().putLong("last", total).putInt(key, now).apply()
                  return now
              }
          }

          package com.nadiayousuf.stepbyn

          import android.Manifest
          import android.app.Activity
          import android.content.Context
          import android.content.Intent
          import android.content.pm.PackageManager
          import android.hardware.Sensor
          import android.hardware.SensorManager
          import android.os.Build
          import android.os.Bundle
          import android.webkit.JavascriptInterface
          import android.webkit.WebView

          class MainActivity : Activity() {
              private lateinit var web: WebView

              override fun onCreate(savedInstanceState: Bundle?) {
                  super.onCreate(savedInstanceState)
                  web = WebView(this)
                  setContentView(web)
                  web.settings.javaScriptEnabled = true
                  web.settings.domStorageEnabled = true
                  web.addJavascriptInterface(Bridge(applicationContext), "AndroidSteps")
                  web.loadUrl("file:///android_asset/index.html")
                  askPermissions()
              }

              override fun onResume() { super.onResume(); startCounting() }

              override fun onBackPressed() { moveTaskToBack(true) }

              private fun askPermissions() {
                  val need = mutableListOf<String>()
                  if (Build.VERSION.SDK_INT >= 29 && checkSelfPermission(Manifest.permission.ACTIVITY_RECOGNITION) != PackageManager.PERMISSION_GRANTED) need.add(Manifest.permission.ACTIVITY_RECOGNITION)
                  if (Build.VERSION.SDK_INT >= 33 && checkSelfPermission(Manifest.permission.POST_NOTIFICATIONS) != PackageManager.PERMISSION_GRANTED) need.add(Manifest.permission.POST_NOTIFICATIONS)
                  if (need.isNotEmpty()) requestPermissions(need.toTypedArray(), 1) else startCounting()
              }

              override fun onRequestPermissionsResult(requestCode: Int, permissions: Array<out String>, grantResults: IntArray) {
                  super.onRequestPermissionsResult(requestCode, permissions, grantResults)
                  startCounting()
              }

              private fun startCounting() {
                  if (Build.VERSION.SDK_INT < 29 || checkSelfPermission(Manifest.permission.ACTIVITY_RECOGNITION) == PackageManager.PERMISSION_GRANTED) {
                      try { startForegroundService(Intent(this, StepService::class.java)) } catch (e: Exception) { }
                  }
              }
          }

          /** Methods the web page calls as AndroidSteps.today(), .week(), .status() */
          class Bridge(private val c: Context) {
              @JavascriptInterface fun today(): Int = Store.steps(c)
              @JavascriptInterface fun week(): String = Store.lastSevenDays(c).joinToString(",", "[", "]") { it.second.toString() }
              @JavascriptInterface fun status(): String {
                  val sm = c.getSystemService(Context.SENSOR_SERVICE) as SensorManager
                  if (sm.getDefaultSensor(Sensor.TYPE_STEP_COUNTER) == null) return "No step sensor"
                  if (Build.VERSION.SDK_INT >= 29 && c.checkSelfPermission(Manifest.permission.ACTIVITY_RECOGNITION) != PackageManager.PERMISSION_GRANTED) return "Permission needed"
                  return "Counting ON"
              }
          }
          SBN_EOF
